"""Read-only saved-source rays; no source saving, rendering or native input."""
from pathlib import Path
import bpy, hashlib, json, math
from mathutils import Vector
root=Path.cwd()
out=[]
for family,filename,manifestname in [('table','furniture.canteen.dining-table.soft-light.blend','oblique-furniture.canteen-dining-table.v1.json'),('bench','furniture.corridor.bench.soft-light.blend','oblique-canteen-bench.v1.json')]:
 path=root/'assets/source/blender'/filename
 before=hashlib.sha256(path.read_bytes()).hexdigest()
 manifest=json.loads((root/'public/game-content'/manifestname).read_text())
 assert before==manifest['sourceSha256']
 bpy.ops.wm.open_mainfile(filepath=str(path));scene=bpy.context.scene;camera=scene.camera
 target=Vector(manifest['cameraTargetTiles']);poses=[]
 for yaw in (30,60,300) if family=='table' else range(0,360,30):
  azimuth,tilt=math.radians(yaw),math.radians(40)
  camera.location=target+Vector((6*math.cos(tilt)*math.sin(azimuth),-6*math.cos(tilt)*math.cos(azimuth),6*math.sin(tilt)))
  camera.rotation_euler=(target-camera.location).to_track_quat('-Z','Y').to_euler()
  bpy.context.view_layer.update();graph=bpy.context.evaluated_depsgraph_get();quat=camera.matrix_world.to_quaternion();direction=quat@Vector((0,0,-1));labels=[]
  for y in range(256):
   for x in range(256):
    origin=camera.location+quat@Vector(((x+.5-128)/64,(128-y-.5)/64,0))
    hit,location,normal,face,obj,matrix=scene.ray_cast(graph,origin,direction,distance=20)
    labels.append(obj.name if hit else None)
  rows=[]
  for name in sorted(set(labels)-{None}):
   wanted=(name.startswith('Enamel plate rim.') or name.startswith('Recessed plate well.')) if family=='table' else (name.startswith('Worn timber slat.') or name.startswith('Crowned wood edge.'))
   if not wanted:continue
   hits=[i for i,m in enumerate(labels) if m==name]
   inset=[i for i in hits if 1<=i%256<=254 and 1<=i//256<=254 and all(labels[i+dy*256+dx]==name for dx in (-1,0,1) for dy in (-1,0,1))]
   rows.append({'mesh':name,'materials':[m.name for m in bpy.data.objects[name].data.materials],'rayHitPixelIndices':hits,'sameMesh3x3InsetPixelIndices':inset})
  poses.append({'yawDegrees':yaw,'elevationDegrees':40,'meshes':rows})
 assert before==hashlib.sha256(path.read_bytes()).hexdigest()
 out.append({'family':family,'source':str(path.relative_to(root)),'sourceSha256':before,'descriptor':manifestname,'sourceUnchanged':True,'renderPerformed':False,'sourceEdited':False,'poses':poses})
 print(family,[(p['yawDegrees'],sum(len(m['rayHitPixelIndices']) for m in p['meshes'])) for p in poses],flush=True)
(root/'docs/research/2026-10-04-canteen-bench-source-material-observer/material-rays.json').write_text(json.dumps({'blenderVersion':list(bpy.app.version),'resolution':[256,256],'pixelsPerTile':64,'families':out},indent=2)+'\n')
