const path = require('node:path');
const fs = require('node:fs');
const root = process.cwd();
const {createRequire}=require('node:module');
const fromRoot=createRequire(path.join(root,'package.json'));
const esbuild=createRequire(fromRoot.resolve('vite/package.json'))('esbuild');
const scratch = path.join(process.env.TEMP, 'lockstate-public-hired-guard-prep-20261003');
const out = path.join(scratch, 'genuine-packed-guard.cjs');
const source = `
import assert from 'node:assert/strict';
import {createNewSimulationRuntime} from './src/simulation/runtime/new-session';
import {packCommand} from './src/simulation/protocol/commands';
import {captureSessionSnapshot,restoreSimulationRuntime} from './src/simulation/runtime/restore-session';
import {actorsFromSnapshot} from './src/rendering/feed/actors-from-snapshot';
const runtime=createNewSimulationRuntime(0);
const empty=captureSessionSnapshot(runtime);
const command={type:'HireStaff',staffRoleId:'staff-role.guard',x:16,y:16} as const;
runtime.kernel.submitCommand('actual-public-hire-source-control',runtime.kernel.expectedSequence,runtime.kernel.tick,packCommand(command));
runtime.kernel.step();
const snapshot=captureSessionSnapshot(runtime);
const actual=JSON.parse(JSON.stringify(snapshot));
const restored=restoreSimulationRuntime(actual).runtime;
assert.equal(runtime.refusals.count,0);
assert.deepEqual(actual.simulation.security.guards.records,[[0,{staffRoleId:'staff-role.guard',tileX:16,tileY:16,deploymentPhase:'unassigned'}]]);
assert.deepEqual(actual.simulation.inFlight.guards.locomotion,{walks:[],headings:[]});
const actors=actorsFromSnapshot(snapshot.simulation,snapshot.entities);
assert.deepEqual(actors,[{id:2**32,assetId:'actor.guard.base',tileX:16,tileY:16,deltaX:0,deltaY:0}]);
assert.deepEqual(JSON.parse(JSON.stringify(captureSessionSnapshot(restored))),actual);
console.log(JSON.stringify({scope:'real packed kernel command and JSON transport snapshot restore; NOT native UI or encoded V8 codec proof',seed:0,command,initialGuards:empty.simulation?.security.guards.records,guards:actual.simulation.security.guards,identity:actual.identity,locomotion:actual.simulation.inFlight.guards.locomotion,derivedActors:actors,wholeTransportSnapshotRoundTripEqual:true},null,2));
`;
esbuild.buildSync({stdin:{contents:source,resolveDir:root,sourcefile:'genuine-packed-guard.ts',loader:'ts'},platform:'node',format:'cjs',bundle:true,outfile:out,logLevel:'warning'});
require(out);
