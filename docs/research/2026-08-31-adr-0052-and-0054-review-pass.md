## 0. What the condition actually says

The brief that commissioned this pass described the condition as a review pass
that had never been done. **That description is right**, and it is right in the
owner's own words rather than by paraphrase.

[Issue #535](https://github.com/matmaxalez/lockstate/issues/535), *"[Owner
decisions] Eight calls taken 2026-08-29 — the player-facing queue is cleared"*,
decision 8, read from the issue itself:

> ## 8. ADR 0051, 0052, 0054 — accept all three
>
> **Decision: accept, with a review pass first.** All three read `Proposed,
> 2026-08-28. Not self-approved.` while the code they describe is live and
> unchallenged.
>
> The owner's instruction was explicit: accept them, **but report back rather
> than quietly approving any ADR that describes something the code does not
> actually do.** A status is supposed to describe reality; that cuts both ways.

So the condition has three parts, and all three shape this document:

1. **A review pass comes first.** The acceptance is not self-executing.
2. **The output is a report, not a status edit.** "Report back rather than
   quietly approving."
3. **It cuts both ways.** An ADR describing something the code does not do is
   the failure the condition names; so is a status that understates a decision
   the code already keeps.

`docs/adr/STATUS-QUEUE.md` records the same condition, quoting it, and adds the
part the issue does not: that the acceptance was discharged for **0051 only**,
by a different ruling ([#639](https://github.com/matmaxalez/lockstate/issues/639)),
and that nothing tracks the remaining two except that paragraph —

> **The earlier ruling was conditional, and the condition is why the remaining
> two are not a debt anybody may simply pay.** … So moving 0052 or 0054 requires
> measuring the document against `main` first and reporting, not editing a status
> line — and `AGENTS.md`'s *"never self-approve"* means the report goes to the
> owner either way. **#639 supplied exactly that for 0051** by naming the ADR as
> *"a decision the code already keeps, waiting on a signature"*; nothing has yet
> supplied it for the other two.

[Issue #703](https://github.com/matmaxalez/lockstate/issues/703) restates it
three times — in its body's *Still outstanding* and again in two of its three
comments — the last as recently as 13:34 UTC on 2026-08-31: *"ADRs 0052 and 0054
remain `Proposed` under #535 decision 8's *conditional* acceptance. Neither has

**One correction to the brief.** It described #703 as carrying *"the owner's

sheets with an independent PNG decoder (zlib + unfilter; no library in this
container reads PNGs) and applying the ADR's own rule — *"the bounding box is

## Appendix — the probe §2.4 was measured with

Kept here in full so §2.4 is reproducible after this session ends, which is the
one thing ADR 0054's own probe did not do. It was run from
`<worktree>/.probe/adr0054.probe.test.ts` beside a config whose `include` is
`['.probe/**/*.probe.test.ts']` and whose `testTimeout` is `600_000`, then
deleted from the worktree before committing — a probe is not a test and does not
belong under `tests/`. Six runs, 761 ms.

```ts
import { appendFileSync } from 'node:fs';
import { describe, expect, it } from 'vitest';
import { DEFAULT_ACTIONS } from '../src/simulation/prisoners/actions';
import { NEED_IDS, NEED_MAX, NEED_SCALE, type NeedId } from '../src/simulation/prisoners/needs';
import { DAY_LENGTH_TICKS } from '../src/simulation/prisoners/regime';
import { packCommand } from '../src/simulation/protocol/commands';
import { createNewSimulationRuntime, type SimulationRuntime } from '../src/simulation/runtime/new-session';
import { wallRoomPerimeter } from '../tests/helpers/room-walls';

const SEED = 0x0054;
const ARRIVAL = { x: 16, y: 16 } as const;             // src/main.ts's NEW_PRISON_ORIGIN_TILE
const ADMISSION = { sentenceLengthTicks: 400_000, priorIncidents: 0 } as const;
const SHOWER = { x: 26, y: 1, width: 3, height: 3 } as const;
const YARD = { x: 20, y: 20, width: 8, height: 8 } as const;
const LAUNDRY = { x: 10, y: 10, width: 3, height: 3 } as const;
const SECTOR = 'security-sector.prison';
const RUN_UNTIL = 24_000;   // ten in-game days at DAY_LENGTH_TICKS 2,400
const WATCH_FROM = 2_000;   // past intake and every build order

function submit(runtime: SimulationRuntime, id: string, payload: ReturnType<typeof packCommand>): void {
  runtime.kernel.submitCommand(id, runtime.kernel.expectedSequence, runtime.kernel.tick, payload);
  runtime.kernel.step();
}
function stepTo(runtime: SimulationRuntime, tick: number): void {
  while (runtime.kernel.tick < tick) runtime.kernel.step();
}
function cellRect(index: number) {
  return { x: 1 + index * 3, y: 1, width: 2, height: 3 };
}

interface Plan {
  readonly prisoners: number;
  readonly guards: number;
  readonly shower: boolean;
  readonly yard: boolean;
  readonly laundry: boolean;
}

function build(plan: Plan): SimulationRuntime {
  const runtime = createNewSimulationRuntime(SEED);
  const cells = Array.from({ length: plan.prisoners }, (_u, i) => cellRect(i));
  submit(runtime, 'buy-planks', packCommand({ type: 'PurchaseMaterials', orderId: 'buy-p', itemId: 'item.wood-plank', quantity: plan.prisoners }));
  submit(runtime, 'buy-bricks', packCommand({ type: 'PurchaseMaterials', orderId: 'buy-b', itemId: 'item.brick', quantity: plan.prisoners + 6 }));
  for (const rect of cells) wallRoomPerimeter(runtime.world, rect, { doors: runtime.navigation.doors });
  if (plan.shower) wallRoomPerimeter(runtime.world, SHOWER, { doors: runtime.navigation.doors });
  if (plan.yard) wallRoomPerimeter(runtime.world, YARD, { doors: runtime.navigation.doors });
  if (plan.laundry) wallRoomPerimeter(runtime.world, LAUNDRY, { doors: runtime.navigation.doors });
  cells.forEach((rect, i) => submit(runtime, `zone-cell-${String(i)}`, packCommand({ type: 'ZoneRoom', roomId: 'room.cell', ...rect })));
  if (plan.shower) submit(runtime, 'zone-shower', packCommand({ type: 'ZoneRoom', roomId: 'room.shower-room', ...SHOWER }));
  if (plan.yard) submit(runtime, 'zone-yard', packCommand({ type: 'ZoneRoom', roomId: 'room.yard', ...YARD }));
  if (plan.laundry) submit(runtime, 'zone-laundry', packCommand({ type: 'ZoneRoom', roomId: 'room.laundry', ...LAUNDRY }));
  cells.forEach((rect, i) => {
    submit(runtime, `bed-${String(i)}`, packCommand({ type: 'PlaceObject', orderId: `bed-${String(i)}`, definitionId: 'bed-wooden', x: rect.x, y: rect.y }));
    submit(runtime, `wc-${String(i)}`, packCommand({ type: 'PlaceObject', orderId: `wc-${String(i)}`, definitionId: 'toilet-brick', x: rect.x + 1, y: rect.y }));
  });
  if (plan.shower) {
    submit(runtime, 'head-1', packCommand({ type: 'PlaceObject', orderId: 'head-1', definitionId: 'shower-head-brick', x: SHOWER.x, y: SHOWER.y }));
    submit(runtime, 'head-2', packCommand({ type: 'PlaceObject', orderId: 'head-2', definitionId: 'shower-head-brick', x: SHOWER.x + 1, y: SHOWER.y }));
  }
  if (plan.laundry) {
    submit(runtime, 'wm-1', packCommand({ type: 'PlaceObject', orderId: 'wm-1', definitionId: 'washing-machine-brick', x: LAUNDRY.x, y: LAUNDRY.y }));
    submit(runtime, 'wm-2', packCommand({ type: 'PlaceObject', orderId: 'wm-2', definitionId: 'washing-machine-brick', x: LAUNDRY.x, y: LAUNDRY.y + 1 }));
  }
  stepTo(runtime, 1_000);
  for (let i = 0; i < plan.guards; i += 1) submit(runtime, `hire-${String(i)}`, packCommand({ type: 'HireStaff', staffRoleId: 'staff-role.guard', ...ARRIVAL }));
  for (let i = 0; i < plan.prisoners; i += 1) submit(runtime, `admit-${String(i)}`, packCommand({ type: 'AdmitPrisoner', ...ADMISSION, ...ARRIVAL }));
  // A refused command would make every figure a measurement of a different prison.
  expect(runtime.refusals.count, 'the fixture must build the prison it says it builds').toBe(0);
  return runtime;
}

function measure(plan: Plan, label: string): void {
  const runtime = build(plan);
  stepTo(runtime, WATCH_FROM);
  const store = runtime.prisoners.entityStore;
  const ids = Array.from({ length: plan.prisoners }, (_u, i) => store.getIdByIndex(i));
  const idle: number[] = ids.map(() => 0);
  const travelling: number[] = ids.map(() => 0);
  const performingTicks: Record<string, number> = {};
  const lowest: Record<NeedId, number> = {} as Record<NeedId, number>;
  const highest: Record<NeedId, number> = {} as Record<NeedId, number>;
  for (const need of NEED_IDS) { lowest[need] = NEED_MAX; highest[need] = 0; }
  let peakScore = 0;
  let finalScore = 0;
  const ticks = RUN_UNTIL - WATCH_FROM;
  for (let tick = runtime.kernel.tick + 1; tick <= RUN_UNTIL; tick += 1) {
    stepTo(runtime, tick);
    ids.forEach((entityId, slot) => {
      const index = store.getIndex(entityId);
      const phase = runtime.prisoners.currentAction.phase[index];   // ACTION_PHASES: 0 idle, 1 travelling, 2 performing
      const actionIndex = runtime.prisoners.currentAction.actionIndex[index]!;
      if (phase === 0) idle[slot]! += 1;
      if (phase === 1) travelling[slot]! += 1;
      if (phase === 2 && actionIndex >= 0) {
        const id = DEFAULT_ACTIONS[actionIndex]!.id;
        performingTicks[id] = (performingTicks[id] ?? 0) + 1;
      }
      if (slot === 0) {
        for (const need of NEED_IDS) {
          // The stored Uint16Array, divided rather than read through the rounding accessor.
          const level = runtime.prisoners.needs.levels[need][index]! / NEED_SCALE;
          if (level < lowest[need]) lowest[need] = level;
          if (level > highest[need]) highest[need] = level;
        }
      }
    });
    finalScore = runtime.sectorRisk.getScore(SECTOR);
    if (finalScore > peakScore) peakScore = finalScore;
  }
  const perDay = (n: number) => (n / ticks) * DAY_LENGTH_TICKS;
  const metrics = runtime.prisoners.actionSystem.getMetrics();
  appendFileSync('probe-out.jsonl', JSON.stringify({
    label, plan, windowTicks: ticks,
    idlePerDay: idle.map(perDay).map((v) => Math.round(v)),
    travellingPerDay: travelling.map(perDay).map((v) => Math.round(v)),
    unmetDemandCycles: metrics.unmetDemandCycles,
    routeFailures: metrics.routeFailures,
    actionsStarted: metrics.actionsStarted,
    performingTicks,
    lowestHygiene: lowest.hygiene, highestHygiene: highest.hygiene,
    lowestRecreation: lowest.recreation, highestRecreation: highest.recreation,
    lowestSafety: lowest.safety,
    peakScore: Number(peakScore.toFixed(4)),
    finalScore: Number(finalScore.toFixed(4)),
    riots: runtime.incidentTriggerSystem.getMetrics().riotsTriggered,
  }) + '\n');
}

describe('ADR 0054 consequences table, re-measured', () => {
  it('cells only, 1 prisoner, 1 guard', () => { measure({ prisoners: 1, guards: 1, shower: false, yard: false, laundry: false }, 'cells only'); });
  it('cells only, 4 prisoners, 1 guard', () => { measure({ prisoners: 4, guards: 1, shower: false, yard: false, laundry: false }, 'cells only, 4 prisoners'); });
  it('cells + yard', () => { measure({ prisoners: 1, guards: 1, shower: false, yard: true, laundry: false }, 'cells + yard'); });
  it('cells + laundry', () => { measure({ prisoners: 1, guards: 1, shower: false, yard: false, laundry: true }, 'cells + laundry'); });
  it('cells + yard + shower + laundry', () => { measure({ prisoners: 1, guards: 1, shower: true, yard: true, laundry: true }, 'cells + yard + shower + laundry'); });
  it('cells only, 1 prisoner, 0 guards', () => { measure({ prisoners: 1, guards: 0, shower: false, yard: false, laundry: false }, 'cells only, no guard'); });
});
```
