it('the actual #531 browser consumer setup agrees with real canonical chooser eligibility', async () => {

 const start=source.indexOf(label);expect(start).toBeGreaterThan(-1);

 let selected='wall-brick',edge='north',collapsed='true';

 const locator=(selector:string):any=>({selector,locator:(child:string)=>locator(`${selector} ${child}`),click:async()=>{const id=/data-buildable="([^"]+)"/.exec(selector)?.[1];if(id)selected=id; const choice=/data-choice="([^"]+)"/.exec(selector)?.[1];if(choice){expect(visible(),'actual selected row must expose a pressable edge').toBe(true);edge=choice;}if(selector.includes('ui-section__header'))collapsed='false';},count:()=>{const id=/data-buildable="([^"]+)"/.exec(selector)?.[1];return id===undefined||id===selected?1:0;},attribute:(name:string)=>name==='data-collapsed'?collapsed:name==='aria-checked'?String(selector.includes(`data-choice="${edge}"`)):undefined,visible});

 const AsyncFunction=Object.getPrototypeOf(async()=>{}).constructor;
 await new AsyncFunction('page','expect','openApp',body)(page,assertion,async()=>{});
 expect(selected).toBe('door-wooden');expect(edge).toBe('west');expect(visible()).toBe(true);
});
