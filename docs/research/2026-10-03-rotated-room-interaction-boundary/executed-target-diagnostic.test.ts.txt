
import assert from 'node:assert/strict';
import {expect,it} from 'vitest';
import {createNewSimulationRuntime} from '../../src/simulation/runtime/new-session';
import {captureSessionSnapshot,restoreSimulationRuntime} from '../../src/simulation/runtime/restore-session';
import {createSaveEnvelope,decodeSaveEnvelope} from '../../src/persistence/save-schema';
import {packCommand} from '../../src/simulation/protocol/commands';
import {routeWaypoints} from '../../src/simulation/navigation/route';
import {DEFAULT_ACTIONS} from '../../src/simulation/prisoners/actions';
function measure(wallOffset:number,loadAfterWall:boolean){
const results:any[]=[];
function send(r:ReturnType<typeof createNewSimulationRuntime>,c:Parameters<typeof packCommand>[0]){const seq=r.kernel.expectedSequence;r.kernel.submitCommand('interaction-'+seq,seq,r.kernel.tick,packCommand(c));assert.equal(r.kernel.dispatchDueCommands(),1);}
function until(r:ReturnType<typeof createNewSimulationRuntime>,p:()=>boolean,label:string){let n=0;for(;n<40000&&!p();n++)r.kernel.step();assert(p(),label+': '+JSON.stringify(r.construction.allOrders().filter(o=>o.state!=='completed').map(o=>({id:o.id,state:o.state,failReason:o.failReason}))));return n;}
function reload(r:ReturnType<typeof createNewSimulationRuntime>){const bundle=captureSessionSnapshot(r);const envelope=createSaveEnvelope({...bundle,gameVersion:'audit',prisonId:'interaction',revision:1,createdAt:0,updatedAt:1});const decoded=decodeSaveEnvelope(JSON.parse(JSON.stringify(envelope)));assert(decoded.ok,'actual save must decode');assert.equal(decoded.value.saveSchemaVersion,8);return restoreSimulationRuntime(decoded.value.payload).runtime;}
const physical=[{templateId:'canteen-basic',roomId:'room.canteen',origin:{x:20,y:5},width:8,height:8,objects:[{id:'object.dining-table',x:1,y:1,w:3,h:2},{id:'object.dining-table',x:4,y:1,w:3,h:2}]},{templateId:'laundry-basic',roomId:'room.laundry',origin:{x:20,y:16},width:6,height:6,objects:[{id:'object.washing-machine',x:1,y:1,w:2,h:1},{id:'object.washing-machine',x:3,y:1,w:2,h:1}]},{templateId:'cell-basic',roomId:'room.cell',origin:{x:5,y:20},width:4,height:7,objects:[{id:'object.bed',x:1,y:1,w:1,h:2},{id:'object.toilet',x:2,y:4,w:1,h:1}]}];
function independentPoint(t,x,y,q,m){const mx=m?t.width-1-x:x;const p=[[mx,y],[t.height-1-y,mx],[t.width-1-mx,t.height-1-y],[y,t.width-1-mx]][q];return{x:t.origin.x+p[0],y:t.origin.y+p[1]};}
function footprint(t,o,q,m){const tiles=[];for(let dy=0;dy<o.h;dy++)for(let dx=0;dx<o.w;dx++)tiles.push(independentPoint(t,o.x+dx,o.y+dy,q,m));return tiles;}
function inspect(r,q,m,phase){const rooms=[];for(const t of physical){const instance=r.prisoners.roomInstances.allByRoomCatalogId(t.roomId)[0];assert(instance,t.roomId+' completed registry');const objectChecks=[];for(const o of t.objects){const tiles=footprint(t,o,q,m);const anchor={x:Math.min(...tiles.map(t=>t.x)),y:Math.min(...tiles.map(t=>t.y))};const actual=r.placedObjects.getSnapshot().find(p=>p.objectId===o.id&&p.anchorTile.x===anchor.x&&p.anchorTile.y===anchor.y);assert(actual,'independent full footprint anchor '+JSON.stringify({t:t.templateId,o:o.id,anchor}));assert.equal(actual.orientation,q);assert(actual.sourceOrderId);for(const tile of tiles){assert.equal(r.world.getSquareStructure(tile),0,'fixture footprint square');assert(r.placedObjects.isTileOccupied(tile),'actual registry claims full footprint');}objectChecks.push({objectId:o.id,anchor,tiles,sourceOrderId:actual.sourceOrderId});}
if(t.roomId==='room.canteen')assert.equal(r.prisoners.roomInstances.concurrentUseCapacityFor(instance,'dining'),6);if(t.roomId==='room.laundry')assert.equal(r.prisoners.roomInstances.concurrentUseCapacityFor(instance,'laundry'),4);if(t.roomId==='room.cell')assert.equal(instance.residentCapacity,1);const from={x:16,y:16},to=instance.anchorTile;assert.equal(r.world.getSquareStructure(to),0,'action destination is not a wall');const id=phase+'-'+t.roomId+'-'+r.kernel.tick;r.navigation.requestRoute(id,from,to,{role:'prisoner',securityClearance:0},0,r.kernel.tick);until(r,()=>r.navigation.getResult(id)!==undefined,'actual path query');const route=r.navigation.getResult(id).result;assert(route.ok,'room anchor route '+JSON.stringify(route));const points=routeWaypoints(route.route);assert.deepEqual(points.at(-1),to);for(const point of points)assert.notEqual(r.world.getSquareStructure(point),1,'path does not cross square wall');assert(route.route.segments.some(s=>s.enteredViaDoorId!==undefined));r.navigation.clearResult(id);rooms.push({roomId:t.roomId,instanceId:instance.instanceId,actualActionDestination:to,physicalObjectAtDestination:r.placedObjects.objectAt(to)?.objectId,objects:objectChecks,capabilities:instance.concurrentUseCapacityByCapability,routeWaypointCount:points.length,routeCost:route.route.totalCost});}return rooms;}
for(const [q,m]of [[1,false]]){let r=createNewSimulationRuntime(73);send(r,{type:'PlaceRoomTemplate',templateId:'storage-room-basic',origin:{x:5,y:5}});send(r,{type:'PlaceRoomTemplate',templateId:'delivery-bay-basic',origin:{x:12,y:5}});const bootstrapTicks=until(r,()=>r.roomTemplates.snapshot().pending.length===0&&r.construction.allOrders().every(o=>o.state==='completed'),'legal capacity bootstrap');for(const t of physical)send(r,{type:'PlaceRoomTemplate',templateId:t.templateId,origin:t.origin,quarterTurns:q,mirrorX:m});const completionTicks=until(r,()=>r.roomTemplates.snapshot().pending.length===0&&r.construction.allOrders().every(o=>o.state==='completed'),'actual rotated completion');const live=inspect(r,q,m,'live');r=reload(r);const loaded=inspect(r,q,m,'V8');assert.deepEqual(loaded.map(x=>({roomId:x.roomId,objects:x.objects,capabilities:x.capabilities,actualActionDestination:x.actualActionDestination})),live.map(x=>({roomId:x.roomId,objects:x.objects,capabilities:x.capabilities,actualActionDestination:x.actualActionDestination})));
send(r,{type:'HireStaff',staffRoleId:'staff-role.guard',x:16,y:16});send(r,{type:'AdmitPrisoner',sentenceLengthTicks:10000,priorIncidents:0,x:16,y:16});until(r,()=>r.prisoners.roomInstances.totalOccupancy===1,'genuine staged intake');assert.equal(r.prisoners.entityStore.isIndexAlive(0),true);const actions=[];for(const categories of [['meal'],['work'],['sleep'],['hygiene']]){for(const b of r.prisoners.regimes.all().find(x=>x.classificationGroupId==='general-population').blocks)send(r,{type:'EditRegimeBlock',classificationGroupId:'general-population',startTickOfDay:b.startTickOfDay,allowedCategories:categories});const seen={};for(let n=0;n<1500;n++){r.kernel.step();const index=r.prisoners.currentAction.actionIndex[0];if(index>=0&&r.prisoners.currentAction.phase[0]===2){const a=DEFAULT_ACTIONS[index];const target=r.prisoners.coldState.getActionTarget(r.prisoners.entityStore.getIdByIndex(0));if(!seen[a.id])seen[a.id]={id:a.id,target,position:{x:r.prisoners.position.tileX[0],y:r.prisoners.position.tileY[0]}};}}actions.push({categories,performing:Object.values(seen)});}const cant=r.prisoners.roomInstances.allByRoomCatalogId('room.canteen')[0];assert.equal(r.placedObjects.isTileOccupied(cant.anchorTile),false);for(const b of r.prisoners.regimes.all().find(x=>x.classificationGroupId==='general-population').blocks)send(r,{type:'EditRegimeBlock',classificationGroupId:'general-population',startTickOfDay:b.startTickOfDay,allowedCategories:['work','meal','sleep','hygiene']});const beforeWallFunds=r.treasury.balanceMinorUnits;send(r,{type:'PlaceBuildOrder',definitionId:'wall-brick',orderId:'actual-later-canteen-anchor-wall',x:cant.anchorTile.x,y:cant.anchorTile.y+wallOffset,footprint:'square'});const order=r.construction.getOrder('actual-later-canteen-anchor-wall');if(order.state!=='failed')until(r,()=>order.state==='completed','later wall completion');const routes=[];for(const to of [cant.anchorTile,{x:cant.anchorTile.x,y:cant.anchorTile.y+1}]){const id='after-wall-'+to.y;r.navigation.requestRoute(id,{x:16,y:16},to,{role:'prisoner',securityClearance:0},0,r.kernel.tick);until(r,()=>r.navigation.getResult(id)!==undefined,'later wall path');routes.push({to,result:r.navigation.getResult(id).result});r.navigation.clearResult(id);}for(const b of r.prisoners.regimes.all().find(x=>x.classificationGroupId==='general-population').blocks)send(r,{type:'EditRegimeBlock',classificationGroupId:'general-population',startTickOfDay:b.startTickOfDay,allowedCategories:['meal']});until(r,()=>!r.prisoners.entityStore.isIndexAlive(0),'genuine first sentence departure');send(r,{type:'AdmitPrisoner',sentenceLengthTicks:100000,priorIncidents:0,x:16,y:16});until(r,()=>r.prisoners.roomInstances.totalOccupancy===1,'next genuine intake');assert(r.prisoners.entityStore.isIndexAlive(0));const freshId=r.prisoners.entityStore.getIdByIndex(0);if(loadAfterWall)r=reload(r);const beforeWatchMetrics=r.prisoners.actionSystem.getMetrics();const beforeHunger=r.prisoners.needs.getScaled(0,'hunger');const afterWallActions={};for(let n=0;n<9000;n++){r.kernel.step();const index=r.prisoners.currentAction.actionIndex[0];if(index>=0&&r.prisoners.currentAction.phase[0]===2){const a=DEFAULT_ACTIONS[index];afterWallActions[a.id]={id:a.id,target:r.prisoners.coldState.getActionTarget(r.prisoners.entityStore.getIdByIndex(0)),position:{x:r.prisoners.position.tileX[0],y:r.prisoners.position.tileY[0]}};}}results.push({quarterTurns:q,mirrorX:m,bootstrapTicks,completionTicks,live,loaded,actions,wall:{order,treasuryBefore:beforeWallFunds,treasuryAfter:r.treasury.balanceMinorUnits},routes,roomAfterWall:r.prisoners.roomInstances.getById(cant.instanceId),candidateAfterWall:r.prisoners.roomInstances.findAvailableForUse('room.canteen','dining'),freshId,beforeWatchMetrics,afterWatchMetrics:r.prisoners.actionSystem.getMetrics(),beforeHunger,afterHunger:r.prisoners.needs.getScaled(0,'hunger'),afterWallActions:Object.values(afterWallActions)});}
return results[0];
}

it.each([[0,false],[0,true],[1,false],[1,true]] as const)("completed q1 Canteen remains usable after later interior wall: offset%s V8%s",(offset,load)=>{const actual=measure(offset,load);expect(actual.afterWallActions.map((a:any)=>a.id)).toContain("action.eat-meal");expect(actual.afterHunger).toBeGreaterThan(0);});
