interface Target { kind: 'room-template'; templateId: string; origin: { x: number; y: number }; quarterTurns?: number; mirrorX?: boolean }
interface Reply { id: string; target: Target; verdict: RoomTemplatePreflight }
interface InputReceipt { type: string; trusted: boolean; canvas: boolean; x?: number; y?: number; code?: string }
interface Probe {
  replies: Reply[];
  quotes: { templateId: string; quote: RoomTemplateCostQuote }[];
  inputs: InputReceipt[];
  snapshot(): Promise<SessionSnapshotBundle>;
}

// Passive tee plus public read-only snapshot requests. Never submits a build,
// history command or projection; those must arise from real controls below.
async function observe(page: Page): Promise<void> {
  await page.addInitScript(() => {
    const RealWorker = Worker;
    let latest: Worker | undefined;
    const requests = new Map<string, { projectionId: string; target: Target }>();
    const readers = new Map<string, (message: unknown) => void>();
    const probe: Probe = { replies: [], quotes: [], inputs: [], async snapshot() {
      if (latest === undefined) throw Error('Actual worker absent');
      const id = crypto.randomUUID();
      const message = await new Promise<unknown>((resolve, reject) => {
        const timer = setTimeout(() => { readers.delete(id); reject(Error('Actual snapshot timed out')); }, 20_000);
        readers.set(id, value => { clearTimeout(timer); resolve(value); });
        latest!.postMessage({ protocolVersion: 1, messageId: id, kind: 'simulation/request-snapshot', payload: { reason: 'consistency-check' } });
      }) as { kind: string; payload: { snapshot: { data: SessionSnapshotBundle } } };
      if (message.kind !== 'simulation/snapshot') throw Error(`Actual snapshot refused: ${message.kind}`);
      return message.payload.snapshot.data;
    } };
    class ObservedWorker extends RealWorker {
      constructor(url: string | URL, options?: WorkerOptions) {
        super(url, options);
        super.addEventListener('message', (event: MessageEvent) => {
          const message = event.data as { replyTo?: string; kind?: string; payload?: { view?: { data?: unknown } } };
          if (message.kind?.startsWith('simulation/')) latest=this;
          const id = message.replyTo ?? '';
          const reader = readers.get(id);
          if (reader !== undefined) { readers.delete(id); reader(event.data); }
          const request = requests.get(id);
          if (message.kind !== 'simulation/projection' || request === undefined || message.payload?.view?.data === undefined) return;
          requests.delete(id);
          if (request.projectionId === 'world/room-template-preflight') probe.replies.push({ id, target: request.target, verdict: message.payload.view.data as RoomTemplatePreflight });
          if (request.projectionId === 'world/room-template-cost') probe.quotes.push({ templateId: request.target.templateId, quote: message.payload.view.data as RoomTemplateCostQuote });
        });
      }
      override postMessage(message: unknown, transfer?: Transferable[] | StructuredSerializeOptions): void {
        const m = message as { messageId?: string; kind?: string; payload?: { projectionId?: string; target?: Target } };
        if (m.kind === 'simulation/request-projection' && m.messageId !== undefined && m.payload?.target !== undefined && (m.payload.projectionId === 'world/room-template-preflight' || m.payload.projectionId === 'world/room-template-cost')) requests.set(m.messageId, { projectionId: m.payload.projectionId, target: m.payload.target });
        if (transfer === undefined) super.postMessage(message);
        else if (Array.isArray(transfer)) super.postMessage(message, transfer);
        else super.postMessage(message, transfer);
      }
    }
    Reflect.set(window, 'Worker', ObservedWorker); Reflect.set(window, 'historyVerdictProbe', probe);
    for (const type of ['pointermove', 'pointerdown', 'pointerup', 'keydown', 'keyup']) window.addEventListener(type, event => {
      if (event instanceof PointerEvent) probe.inputs.push({ type, trusted: event.isTrusted, canvas: event.target === document.querySelector('#game-root canvas'), x: event.clientX, y: event.clientY });
      else if (event instanceof KeyboardEvent) probe.inputs.push({ type, trusted: event.isTrusted, canvas: false, code: event.code });
    }, true);
  });
}
const snapshot = (page: Page) => page.evaluate(() => (Reflect.get(window, 'historyVerdictProbe') as Probe).snapshot());
const replies = (page: Page) => page.evaluate(() => (Reflect.get(window, 'historyVerdictProbe') as Probe).replies);
const inputs = (page: Page) => page.evaluate(() => (Reflect.get(window, 'historyVerdictProbe') as Probe).inputs);
const submitted = (page: Page) => sentCommands(page);
const cellOrders = (data: SessionSnapshotBundle) => data.construction.orders.filter(order => order.location.x >= 20 && order.location.x < 27 && order.location.y >= 5 && order.location.y < 9);
const cellObjects = (data: SessionSnapshotBundle) => data.simulation?.objects?.placedObjects.filter(object => object.anchorTile.x >= 20 && object.anchorTile.x < 27 && object.anchorTile.y >= 5 && object.anchorTile.y < 9) ?? [];
const cellRooms = (data: SessionSnapshotBundle) => data.simulation?.prisoners.roomInstanceDefinitions.filter(room => room.roomCatalogId === 'room.cell') ?? [];
function plane(data: SessionSnapshotBundle, x: number, y: number, name: 'squareStructure' | 'zoning'): number {
  const size = data.world.chunkSize;
  const chunk = data.world.chunks.find(c => c.x === Math.floor(x / size) && c.y === Math.floor(y / size));
  if (chunk === undefined || chunk.lifecycle !== 'loaded') throw Error('Actual occupied chunk absent');
  const runs = chunk[name];
  if (runs === undefined) return 0; // Published absent plane encodes all zeroes.
  let at = y % size * size + x % size;
  for (const [value, count] of runs) { if (at < count) return value; at -= count; }
  throw Error('Malformed actual plane RLE');
}
function assertCompletedCell(data: SessionSnapshotBundle): void {
  const orders = cellOrders(data); expect(orders).toHaveLength(20);
  expect(orders.every(order => order.state === 'completed')).toBe(true);
  const expectedShell: string[] = [];
  for (let y = 5; y < 9; y++) for (let x = 20; x < 27; x++) {
    if ((y === 5 || y === 8 || x === 20 || x === 26) && !(x === 20 && y === 6)) expectedShell.push(`wall-brick@${x},${y}`);
  }
  // The visible door occupies square20,6; its east boundary is stored as
  // the neighbouring tile21,6's west edge (the established q1 contract).
  expectedShell.push('door-wooden@21,6');
  expect(orders.filter(order => order.definitionId === 'door-wooden')).toMatchObject([{ location: { x: 21, y: 6, edge: 'west' } }]);
  expect(orders.map(o => `${o.definitionId}@${o.location.x},${o.location.y}`).sort()).toEqual([...expectedShell, 'bed-wooden@24,6', 'toilet-brick@22,7'].sort());
  expect(cellObjects(data).map(o => ({ id: o.objectId, anchor: o.anchorTile, orientation: o.orientation })).sort((a,b) => a.id.localeCompare(b.id))).toEqual([
    { id: 'object.bed', anchor: { x: 24, y: 6 }, orientation: 1 }, { id: 'object.toilet', anchor: { x: 22, y: 7 }, orientation: 1 },
  ]);
  for (const object of cellObjects(data)) expect(orders.find(o => o.id === object.sourceOrderId)).toMatchObject({ state: 'completed', location: object.anchorTile, objectOrientation: 1 });
  expect(cellRooms(data)).toMatchObject([{ roomCatalogId: 'room.cell', anchorTile: { x: 21, y: 6 }, width: 5, height: 2 }]);
  for (let y = 6; y < 8; y++) for (let x = 21; x < 26; x++) expect(plane(data,x,y,'zoning')).toBe(1);
  expect(data.simulation?.roomTemplates?.completed).toEqual(expect.arrayContaining([expect.objectContaining({ templateId: 'cell-basic', origin, quarterTurns: 1 })]));
}

async function finish(page: Page): Promise<void> {
  const panel = page.locator('.hud-build');
  const queued = async () => { const value=await panel.getAttribute('data-queued'); if(value===null) return 0; if(!/^\d+$/u.test(value)) throw Error('Actual queued counter malformed'); return Number(value); };
  let count = await queued();
  while (count > 0) {
    await expect.poll(queued, { message: `actual construction advances from ${count} queued orders` }).toBeLessThan(count);
    count = await queued();
  }
}
async function viewIdentity(page: Page) {
  return page.evaluate(() => ({
    canvas: (()=>{ const canvas=document.querySelector<HTMLCanvasElement>('#game-root canvas'); if(canvas===null) throw Error('Canvas absent'); return {width:canvas.width,height:canvas.height}; })(),
    points: [...document.querySelectorAll<SVGPolygonElement>('.room-template-world-ghost polygon')].map(p => p.getAttribute('points')),
    camera: (() => { const v = document.querySelector<HTMLElement>('.hud-minimap__viewport'); if (v === null) throw Error('Minimap viewport absent'); return { left: v.style.left, top: v.style.top, width: v.style.width, height: v.style.height }; })(),
  }));
}
function assertWholeQ1Footprint(view: Awaited<ReturnType<typeof viewIdentity>>): void {
  expect(view.points).toHaveLength(28);
  const quads = view.points.map(points => {
    if(points===null) throw Error('Actual polygon points absent');
    const parsed=points.trim().split(/\s+/u).map(pair=>pair.split(',').map(Number));
    expect(parsed).toHaveLength(4);
    expect(parsed.every(p=>p.length===2&&p.every(Number.isFinite))).toBe(true);
    return parsed;
  });
  const first=quads[0]!,start=first[0]!;
  const ux={x:first[1]![0]!-start[0]!,y:first[1]![1]!-start[1]!};
  const uy={x:first[3]![0]!-start[0]!,y:first[3]![1]!-start[1]!};
  expect(Math.abs(ux.x*uy.y-ux.y*uy.x)).toBeGreaterThan(1);
  // Literal q1 Cell is7columns/4rows. Independent affine adjacency check;
  // it never supplies the world origin, which must be actual target20,5.
  for(let y=0;y<4;y++) for(let x=0;x<7;x++) for(let corner=0;corner<4;corner++) {
    const actual=quads[y*7+x]![corner]!, base=first[corner]!;
    expect(actual[0]).toBeCloseTo(base[0]!+x*ux.x+y*uy.x,4);
    expect(actual[1]).toBeCloseTo(base[1]!+x*ux.y+y*uy.y,4);
    expect(actual[0]).toBeGreaterThanOrEqual(0); expect(actual[0]).toBeLessThanOrEqual(view.canvas.width);
    expect(actual[1]).toBeGreaterThanOrEqual(0); expect(actual[1]).toBeLessThanOrEqual(view.canvas.height);
  }
}
async function assertNoPointerSince(page: Page, start: number): Promise<void> {
  expect((await inputs(page)).slice(start).filter(input => input.type.startsWith('pointer'))).toEqual([]);
}
