inputs=[]
def frame(kind,yaw):
 d=json.loads((ROOT/f'public/game-content/oblique-furniture.kitchen-{kind}.v1.json').read_text())
 f=next(x for x in d['frames'] if x['yawDegrees']==yaw and x['elevationDegrees']==40)
 raw=(ROOT/('public'+f['image'])).read_bytes()
 assert hashlib.sha256(raw).hexdigest()==f['sha256']
 inputs.append({'image':f['image'],'sha256':f['sha256']})
 return Image.open(ROOT/('public'+f['image'])).convert('RGBA')
def proj(x,y,z,yaw,target):
 a=math.radians(yaw);e=math.radians(40)
 return (128+64*(math.cos(a)*x+math.sin(a)*y),128-64*(-math.sin(a)*math.sin(e)*x+math.cos(a)*math.sin(e)*y+math.cos(e)*(z-target)))
def palettepoly(im,poly):
 mask=Image.new('1',im.size);ImageDraw.Draw(mask).polygon(poly,fill=1)
 colors=[]
 for y in range(1,255):
  for x in range(1,255):
   if all(mask.getpixel((x+dx,y+dy)) for dx in [-1,0,1] for dy in [-1,0,1]):
    rgb=im.getpixel((x,y))
    if rgb[3]==255:colors.append(rgb[:3])
 return set(colors),colors
