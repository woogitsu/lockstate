const neighbors = [
  'tests/unit/ui-room-template-session-preflight.test.ts',
  'tests/unit/ui-room-template-tool.test.ts',
  'tests/unit/ui-room-template-dialog-status.test.ts',
  'tests/unit/ui-room-template-world-bridge.test.ts',
  'tests/unit/ui-room-template-preview-fit.test.ts',
  'tests/unit/ui-renderer-template-stationary-hover.test.ts',
  'tests/unit/ui-simulation-projections.test.ts',
  'tests/unit/simulation-worker-channel.test.ts',
  'tests/unit/persistence-worker-per-session-host.test.ts',
  'tests/unit/room-template-session.test.ts',
];
const docs = [
  'tests/foundation/research-index-contract.test.ts',
  'tests/foundation/documented-wire-schema-membership-contract.test.ts',
  'tests/foundation/documentation-version-claim-contract.test.ts',

];
const args = mode==='app-ts' ? ['node_modules/typescript/bin/tsc','-b','--pretty','false']
  : mode==='tools-ts' ? ['node_modules/typescript/bin/tsc','-p','tsconfig.tools.json','--pretty','false']
  : mode==='build' ? ['node_modules/vite/bin/vite.js','build']
  : ['node_modules/vitest/vitest.mjs','run',...(mode==='neighbors'?neighbors:docs),'--maxWorkers=2'];
const r = cp.spawnSync(process.execPath,args,{encoding:'utf8',env:{...process.env,GIT_NO_LAZY_FETCH:'1'}});
fs.writeFileSync(`${dir}/${mode}.txt`,r.stdout+r.stderr);
process.stdout.write(r.stdout+r.stderr);
process.stdout.write(`exit=${r.status}\n`);
process.exit(r.status ?? 1);
