import { expect, it } from 'vitest';
import { writeFileSync } from 'node:fs';
import { createNewSimulationRuntime } from '../../src/simulation/runtime/new-session';
import { captureSessionSnapshot, restoreSimulationRuntime } from '../../src/simulation/runtime/restore-session';
import { packCommand } from '../../src/simulation/protocol/commands';
import { createSaveEnvelope, decodeSaveEnvelope } from '../../src/persistence/save-schema';
it('calibrates native prepared literals against genuine packed q1 commands, not a browser execution', () => {
  const runtime = createNewSimulationRuntime(0), initial = captureSessionSnapshot(runtime);
  runtime.kernel.submitCommand('public-plan-shape', 0, 0, packCommand({type:'PlaceRoomTemplate',templateId:'cell-basic',origin:{x:20,y:5},quarterTurns:1}));
  runtime.kernel.dispatchDueCommands(); runtime.kernel.step();
  const pending=captureSessionSnapshot(runtime);
  expect(pending.construction.orders).toHaveLength(18);
  expect(pending.construction.orders.every(order => order.state==='materials-pending')).toBe(true);
  expect(pending.simulation?.economy?.treasury.balanceMinorUnits).toBe(23575);
  expect(pending.construction.newerActionThanTheStackTop).toBe(false);
  const id=pending.construction.orders[0]!.id, revision=pending.construction.orderRevisions![id]!;
  expect(revision).toBe(2);
  const command=packCommand({type:'CancelBuildOrder',orderId:id,expectedRevision:revision});
  runtime.kernel.submitCommand('public-cancel-shape',1,runtime.kernel.tick+12,command);
  const queued=captureSessionSnapshot(runtime);
  const envelope=createSaveEnvelope({gameVersion:'test',prisonId:'public-shape-calibration',revision:1,createdAt:0,updatedAt:1,...queued});
  const decoded=decodeSaveEnvelope(JSON.parse(JSON.stringify(envelope)));
  expect(decoded.ok).toBe(true); if(!decoded.ok)throw Error('genuine V9 data must decode');
  const loaded=restoreSimulationRuntime(decoded.value.payload).runtime;
  expect(captureSessionSnapshot(loaded)).toEqual(queued);
  for(let i=0;i<13;i++)loaded.kernel.step();
  const dispatched=captureSessionSnapshot(loaded);
  expect(dispatched.construction.orders).toEqual(queued.construction.orders.map(order=>({...order,state:'cancelled'})));
  expect(dispatched.construction.undoStack).toEqual(queued.construction.undoStack);
  expect(dispatched.construction.redoStack).toEqual(queued.construction.redoStack);
  expect(dispatched.construction.newerActionThanTheStackTop).toBe(true);
  expect(dispatched.construction.orderRevisions).toEqual(Object.fromEntries(Object.entries(queued.construction.orderRevisions!).map(([id,value])=>[id,value+1])));
  expect(dispatched.simulation?.economy?.treasury.balanceMinorUnits).toBe(25000);
  expect(dispatched.world).toEqual(initial.world);
  expect(dispatched.entities).toEqual(initial.entities);
  expect(dispatched.identity).toEqual(initial.identity);
  expect(dispatched.simulation?.objects?.placedObjects).toEqual([]);
  expect(dispatched.simulation?.roomTemplates?.pending ?? []).toEqual([]);
  writeFileSync('docs/research/2026-10-03-queued-cancellation-v9-native/raw/actual-packed-calibration.json',JSON.stringify({initial,pending,queued,envelope,dispatched},null,2));
});
