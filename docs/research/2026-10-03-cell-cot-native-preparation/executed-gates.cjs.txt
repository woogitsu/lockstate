const fs=require('node:fs'); const {spawnSync}=require('node:child_process');
const dir='docs/research/2026-10-03-cell-cot-native-preparation';
const gates=[['strict-types',['node_modules/typescript/bin/tsc','-b','--pretty','false']],['source-collection',['node_modules/@playwright/test/cli.js','test','--config','tests/browser/playwright.config.ts','cell-cot-player-build.spec.ts','--list']]];
const results=[];for(const [name,args] of gates){const start=Date.now();const r=spawnSync(process.execPath,args,{encoding:'utf8',maxBuffer:32*1024*1024});fs.writeFileSync(`${dir}/${name}.log`,r.stdout+r.stderr);results.push({name,args,exitCode:r.status,elapsedMs:Date.now()-start,error:r.error?.message});if(r.status!==0)break;}
fs.writeFileSync(`${dir}/preparation-gates.json`,JSON.stringify({node:process.version,gates:results,nativeExecuted:false,browserStarted:false,serverStarted:false,productionMutationPerformed:false},null,2));console.log(JSON.stringify(results));
